# Review as data — Implementation Plan (DRAFT)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A published Review is JSON that the canvas app renders; no agent-authored code is ever imported by the app, the desktop server, or the code-oss host.

**Architecture:** Part A turns the per-review software-map bundle from a JS wrapper around JSON into JSON files, and the host fetches JSON instead of blob-importing. Part B replaces the MDX→JSX→esbuild document bundle with a `review-document.json` built at publish from the markdown AST plus one native Node import of `data.ts`; the app renders that tree with the existing component registry. Old sealed revisions are regenerated by `review migrate apply`; a review the migration cannot regenerate shows a republish state.

**Tech Stack:** Node 24 native type stripping and `module.registerHooks`; `mdast-util-from-markdown` + `mdast-util-mdx` + `mdast-util-to-hast` + `property-information`; zod 4; React 19; vitest (`// @vitest-environment jsdom` per file); Hono server; code-oss host under Node's built-in test runner.

**Spec:** `docs/superpowers/specs/2026-09-04-review-as-data-design.md` (same worktree). This plan corrects the spec where the code disagreed; see "Corrections to the spec" below.

```ts
/** JSON-serializable projection of a normalized model. `elementsByPath` is
    derived, so it is rebuilt on load by `hydrateSoftwareModel`. */
export interface SoftwareModelData {
  elements: NormalizedSoftwareElement[];
  relationships: NormalizedSoftwareRelationship[];
}

export function softwareModelData(
  model: NormalizedSoftwareModel,
): SoftwareModelData {
  return { elements: model.elements, relationships: model.relationships };
}

export function hydrateSoftwareModel(
  data: SoftwareModelData,
): NormalizedSoftwareModel {
  return {
    elements: data.elements,
    relationships: data.relationships,
    elementsByPath: new Map(
      data.elements.map((element) => [element.path, element]),
    ),
  };
}
```

```ts
import crypto from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";

import { type JsonValue, parseJsonText } from "@dev.fast/review-protocol";
import { z } from "zod";

import {
  type NormalizedSoftwareModel,
  type SoftwareModelData,
  softwareModelData,
} from "./software-map-model";

export const REVIEW_SOFTWARE_MAP_BUNDLE_DIR = path.join(
  ".bundle",
  "software-map",
);
export const SOFTWARE_MAP_DATA_FORMAT = "software-map/1";
const HEAD_MAP_FILE = "head-map.json";
const BASE_MAP_FILE = "base-map.json";
const MANIFEST_FILE = "manifest.json";
// Version 1 wrote ES modules (head-map.js / base-map.js). Version 2 writes
// JSON; a version-1 bundle reads as null and the review needs republishing.
const MANIFEST_VERSION = 2;

const COMMIT_SHA_PATTERN = /^[0-9a-f]{40}$/i;

const SoftwareMapBundleManifestSchema = z.object({
  version: z.literal(MANIFEST_VERSION),
  headCommit: z.string().regex(COMMIT_SHA_PATTERN),
  baseCommit: z.string().regex(COMMIT_SHA_PATTERN),
});
type SoftwareMapBundleManifest = z.infer<typeof SoftwareMapBundleManifestSchema>;

const SoftwareMapDataFileSchema = z.object({
  format: z.literal(SOFTWARE_MAP_DATA_FORMAT),
  elements: z.array(z.unknown()),
  relationships: z.array(z.unknown()),
});

export interface ReviewSoftwareMapBundle {
  head: SoftwareModelData;
  base: SoftwareModelData;
  headJson: string;
  baseJson: string;
  contentHash: string;
  headCommit: string;
  baseCommit: string;
}

export function bundleReviewSoftwareMap(input: {
  head: NormalizedSoftwareModel;
  base: NormalizedSoftwareModel;
  headCommit: string;
  baseCommit: string;
}): ReviewSoftwareMapBundle {
  const head = softwareModelData(input.head);
  const base = softwareModelData(input.base);
  const headJson = softwareMapDataJson(head);
  const baseJson = softwareMapDataJson(base);
  return {
    head,
    base,
    headJson,
    baseJson,
    contentHash: bundleHash(headJson, baseJson),
    headCommit: input.headCommit,
    baseCommit: input.baseCommit,
  };
}

export async function writeReviewSoftwareMapBundle(
  reviewDir: string,
  bundle: ReviewSoftwareMapBundle,
): Promise<void> {
  const bundleDir = path.join(reviewDir, REVIEW_SOFTWARE_MAP_BUNDLE_DIR);
  await mkdir(bundleDir, { recursive: true, mode: 0o700 });
  const manifest: SoftwareMapBundleManifest = {
    version: MANIFEST_VERSION,
    headCommit: bundle.headCommit,
    baseCommit: bundle.baseCommit,
  };
  await Promise.all([
    writeFile(path.join(bundleDir, HEAD_MAP_FILE), bundle.headJson, "utf8"),
    writeFile(path.join(bundleDir, BASE_MAP_FILE), bundle.baseJson, "utf8"),
    writeFile(
      path.join(bundleDir, MANIFEST_FILE),
      `${JSON.stringify(manifest, null, 2)}\n`,
      "utf8",
    ),
  ]);
}

export async function readReviewSoftwareMapBundle(
  rootDir: string,
): Promise<ReviewSoftwareMapBundle | null> {
  const bundleDir = path.join(rootDir, REVIEW_SOFTWARE_MAP_BUNDLE_DIR);
  let manifestRaw: string;
  try {
    manifestRaw = await readFile(path.join(bundleDir, MANIFEST_FILE), "utf8");
  } catch (error) {
    if (error instanceof Error && "code" in error && error.code === "ENOENT") {
      return null;
    }
    throw error;
  }
  const manifest = parseManifest(manifestRaw);
  if (!manifest) return null;
  const [headJson, baseJson] = await Promise.all([
    readFile(path.join(bundleDir, HEAD_MAP_FILE), "utf8"),
    readFile(path.join(bundleDir, BASE_MAP_FILE), "utf8"),
  ]);
  const head = parseSoftwareMapData(headJson);
  const base = parseSoftwareMapData(baseJson);
  if (!head || !base) return null;
  return {
    head,
    base,
    headJson,
    baseJson,
    contentHash: bundleHash(headJson, baseJson),
    headCommit: manifest.headCommit,
    baseCommit: manifest.baseCommit,
  };
}

export function sameReviewSoftwareMapBundle(
  left: ReviewSoftwareMapBundle,
  right: ReviewSoftwareMapBundle,
): boolean {
  return (
    left.headJson === right.headJson &&
    left.baseJson === right.baseJson &&
    left.headCommit === right.headCommit &&
    left.baseCommit === right.baseCommit
  );
}

function softwareMapDataJson(data: SoftwareModelData): string {
  return `${JSON.stringify({ format: SOFTWARE_MAP_DATA_FORMAT, ...data })}\n`;
}

function parseSoftwareMapData(raw: string): SoftwareModelData | null {
  let value: JsonValue;
  try {
    value = parseJsonText(raw);
  } catch {
    return null;
  }
  const parsed = SoftwareMapDataFileSchema.safeParse(value);
  if (!parsed.success) return null;
  // SAFETY: element/relationship shapes are produced by defineSoftwareMap at
  // publish and re-validated by the app's structural check on load.
  return {
    elements: parsed.data.elements as SoftwareModelData["elements"],
    relationships: parsed.data.relationships as SoftwareModelData["relationships"],
  };
}

function parseManifest(raw: string): SoftwareMapBundleManifest | null {
  let value: JsonValue;
  try {
    value = parseJsonText(raw);
  } catch {
    return null;
  }
  const manifest = SoftwareMapBundleManifestSchema.safeParse(value);
  return manifest.success ? manifest.data : null;
}

function bundleHash(headJson: string, baseJson: string): string {
  return crypto
    .createHash("sha256")
    .update(headJson)
    .update("\0")
    .update(baseJson)
    .digest("hex")
    .slice(0, 20);
}
```

```ts
export type ReviewAuthoringComponentName =
  keyof typeof reviewAuthoringPropsSchemas;

      type: z.literal("component"),
      name: z.enum(componentNames),
      props: z.record(z.string(), z.unknown()),

  ]),

  title: string;
  routePath: string;
  sourcePath: string;

  anchorContents: Record<string, string>;
  softwareModels: SoftwareModelData[];

    title: z.string(),
    routePath: z.string(),
    sourcePath: z.string(),
    body: z.array(reviewNodeSchema),

    peeks against the pinned worktree, then ships pointers only; the app
    resolves on render through /code-peek/resolve. */
export function stripPeekResolutions<T>(value: T): T {
  return JSON.parse(
    JSON.stringify(value, (_key, current: unknown) =>
      current &&
      typeof current === "object" &&
      (current as { __kind?: unknown }).__kind === "code-peek-ref"
        ? { ...(current as object), resolution: null }
        : current,
    ),
  ) as T;
}

  parent: ReviewComponentNode | null = null,
): void {
  for (const node of nodes) {
    visit(node, parent);
    if (node.type === "text") continue;
    walkReviewNodes(node.children, visit, node.type === "component" ? node : parent);
  }
}
```

```ts
import { mkdir, realpath, rm, writeFile } from "node:fs/promises";
import { registerHooks } from "node:module";
import path from "node:path";
import { pathToFileURL } from "node:url";

import type { ReviewDefinitionSession } from "./authoring";

export const REVIEW_AUTHORING_SPECIFIER = "virtual:progressive-review-authoring";
const SESSION_GLOBAL = "__devFastReviewDataSessions";
const MIN_NODE_MAJOR = 24;

export interface ReviewDataModuleResult {
  exports: Record<string, unknown>;
  session: ReviewDefinitionSession;
}

type SessionSlots = Record<string, ReviewDefinitionSession | undefined>;

// data.ts is agent-authored TypeScript. Node 24 strips erasable types
// natively, so the file is imported in place: sibling imports keep working
// and no bundler runs. The one non-file import, the virtual authoring
// module, is resolved by a hook to a per-call shim that hands out this
// publish's definition session. Every URL under the review dir is stamped
// with the call's slot so a second publish in one process never reads Node's
// module cache. This runs agent code in the CLI process — the same trust
// boundary the publish gate always had; the app never imports it.
export async function importReviewDataModule(input: {
  reviewDir: string;
  dataPath?: string;
  session: ReviewDefinitionSession;
}): Promise<ReviewDataModuleResult> {
  const major = Number(process.versions.node.split(".")[0]);
  if (major < MIN_NODE_MAJOR) {
    throw new Error(
      `review publish needs Node ${MIN_NODE_MAJOR} or newer to load data.ts (running ${process.versions.node}).`,
    );
  }
  const slot = `${process.pid}-${Math.random().toString(36).slice(2)}`;
  const reviewDir = await realpath(input.reviewDir);
  const reviewDirUrl = `${pathToFileURL(reviewDir).href}/`;
  const dataPath = path.resolve(reviewDir, input.dataPath ?? "data.ts");
  const shimDir = path.join(reviewDir, ".build", `data-shim-${slot}`);
  const shimPath = path.join(shimDir, "progressive-review-authoring.mjs");
  const authoringUrl = pathToFileURL(
    path.join(path.dirname(new URL(import.meta.url).pathname), "authoring.js"),
  ).href;

  const slots = ((globalThis as Record<string, unknown>)[SESSION_GLOBAL] ??=
    {}) as SessionSlots;
  slots[slot] = input.session;
  await mkdir(shimDir, { recursive: true, mode: 0o700 });
  await writeFile(
    shimPath,
    [
      `const session = globalThis[${JSON.stringify(SESSION_GLOBAL)}][${JSON.stringify(slot)}];`,
      "export const defineActors = session.defineActors;",
      "export const defineAnchors = session.defineAnchors;",
      "export const defineStores = session.defineStores;",
      "export const defineSoftwareActors = session.defineSoftwareActors;",
      "export const defineSoftwareStores = session.defineSoftwareStores;",
      `export { calls, defineSoftwareModel } from ${JSON.stringify(authoringUrl)};`,
      "",
    ].join("\n"),
  );
  const hooks = registerHooks({
    resolve(specifier, context, nextResolve) {
      if (specifier === REVIEW_AUTHORING_SPECIFIER) {
        return { url: pathToFileURL(shimPath).href, shortCircuit: true };
      }
      const resolved = nextResolve(specifier, context);
      if (resolved.url.startsWith(reviewDirUrl) && !resolved.url.includes("?v=")) {
        return { ...resolved, url: `${resolved.url}?v=${slot}` };
      }
      return resolved;
    },
  });
  try {
    const url = `${pathToFileURL(dataPath).href}?v=${slot}`;
    const exports = (await import(url)) as Record<string, unknown>;
    await input.session.ready();
    return { exports: { ...exports }, session: input.session };
  } catch (error) {
    throw new Error(
      `data.ts could not be loaded: ${describeImportError(error)}`,
      { cause: error },
    );
  } finally {
    hooks.deregister();
    delete slots[slot];
    await rm(shimDir, { recursive: true, force: true });
  }
}

function describeImportError(error: unknown): string {
  const message = error instanceof Error ? error.message : String(error);
  const code = (error as { code?: string }).code;
  if (code === "ERR_UNSUPPORTED_TYPESCRIPT_SYNTAX") {
    return `${message}. data.ts must use erasable TypeScript only (no enum, namespace, or parameter properties).`;
  }
  if (code === "ERR_IMPORT_ATTRIBUTE_MISSING") {
    return `${message}. JSON imports in data.ts need \`with { type: "json" }\`.`;
  }
  return message;
}
```

```ts
  message: string;

export interface ReviewDocumentTree {
  title: string;
  body: ReviewNode[];
  components: ReviewComponentNode[];
  errors: ReviewDocumentTreeError[];

const DATA_IMPORT = "./data.ts";
const PASS_THROUGH = [
  "mdxJsxFlowElement",
  "mdxJsxTextElement",
  "mdxjsEsm",
  "mdxFlowExpression",
  "mdxTextExpression",

export function buildReviewDocumentTree(input: {
  source: string;
  bindings: Record<string, unknown>;
}): ReviewDocumentTree {
  const errors: ReviewDocumentTreeError[] = [];
  const source = maskReviewFrontmatter(input.source);
  let mdast: MdastRoot;
  try {
    mdast = fromMarkdown(source, {
      extensions: [mdxjs({ acorn: reviewTypescriptEstreeParser }), gfm()],
      mdastExtensions: [mdxFromMarkdown(), gfmFromMarkdown()],
    });
  } catch (error) {
    return { title: "", body: [], components: [], errors: [{ line: null, message: error instanceof Error ? error.message : String(error) }] };
  }
  const title = documentTitle(mdast);
  const locals = collectImportedLocals(mdast, input.bindings, errors);
  const file = { fail(message: string, node?: unknown): never { throw Object.assign(new Error(message), { reviewLine: nodeLine(node) }); } };
  try {
    remarkReviewAnchorLinks()(mdast, file);
    remarkReviewSections()(mdast);
  } catch (error) {
    errors.push({ line: (error as { reviewLine?: number | null }).reviewLine ?? null, message: error instanceof Error ? error.message : String(error) });
    return { title, body: [], components: [], errors };
  }
  const hast = toHast(mdast, { passThrough: PASS_THROUGH }) as HastRoot;
  rehypeReviewTargets()(hast);
  const components: ReviewComponentNode[] = [];
  const evaluate = expressionEvaluator(locals);
  const body = convertChildren(hast.children as HastNode[], { evaluate, errors, components });
  return { title, body, components, errors };
}

interface ConvertContext {
  evaluate: (expression: string, line: number | null) => unknown;
  errors: ReviewDocumentTreeError[];
  components: ReviewComponentNode[];
}

function convertChildren(nodes: HastNode[], context: ConvertContext): ReviewNode[] {
  const out: ReviewNode[] = [];
  for (const node of nodes) {
    const converted = convertNode(node, context);
    if (converted) out.push(converted);
  }
  return out;
}

function convertNode(node: HastNode, context: ConvertContext): ReviewNode | null {
  switch (node.type as string) {
    case "text":
      return { type: "text", value: (node as { value: string }).value };
    case "element": {
      const element = node as HastElement;
      return {
        type: "element",
        tag: element.tagName,
        props: reactPropsFromHastProperties(element.properties ?? {}),
        children: convertChildren(element.children as HastNode[], context),
      };
    }
    case "mdxjsEsm":
      return null; // consumed by collectImportedLocals
    case "mdxFlowExpression":
    case "mdxTextExpression":
      context.errors.push({ line: nodeLine(node), message: "Bare {expression} blocks are not supported in review prose; put values in component props." });
      return null;
    case "mdxJsxFlowElement":
    case "mdxJsxTextElement":
      return convertJsx(node as unknown as MdxJsxFlowElement | MdxJsxTextElement, context);
    default:
      context.errors.push({ line: nodeLine(node), message: `Unsupported markdown node "${node.type}".` });
      return null;
  }
}

function convertJsx(node: MdxJsxFlowElement | MdxJsxTextElement, context: ConvertContext): ReviewNode | null {
  const line = nodeLine(node);
  if (!node.name) {
    context.errors.push({ line, message: "Fragments (<>…</>) are not supported." });
    return null;
  }
  const raw: Record<string, unknown> = {};
  for (const attribute of node.attributes) {
    if (attribute.type !== "mdxJsxAttribute") {
      context.errors.push({ line, message: `<${node.name}> uses a spread attribute; write each prop explicitly.` });
      return null;
    }
    if (attribute.value == null) raw[attribute.name] = true;
    else if (typeof attribute.value === "string") raw[attribute.name] = attribute.value;
    else {
      try {
        raw[attribute.name] = context.evaluate(attribute.value.value, line);
      } catch (error) {
        context.errors.push({ line, message: `<${node.name} ${attribute.name}={…}>: ${error instanceof Error ? error.message : String(error)}` });
        return null;
      }
    }
  }
  const children = convertChildren(node.children as unknown as HastNode[], context);
  if (/^[a-z]/.test(node.name)) {
    for (const [key, value] of Object.entries(raw)) {
      if (typeof value === "object") {
        context.errors.push({ line, message: `<${node.name}> attribute ${key} must be a literal.` });
        return null;
      }
    }
    return { type: "element", tag: node.name, props: raw as ReviewElementProps, children };
  }
  const name = node.name as ReviewAuthoringComponentName;
  const schema = reviewAuthoringPropsSchemas[name];
  if (!schema) {
    context.errors.push({ line, message: `Unknown component <${node.name}>. Review components are: ${Object.keys(reviewAuthoringPropsSchemas).join(", ")}.` });
    return null;
  }
  const result = schema.safeParse(children.length > 0 ? { ...raw, children } : raw);
  if (!result.success) {
    context.errors.push({ line, message: `<${name}> ${result.error.issues.map((issue) => `${issue.path.join(".") || "props"}: ${issue.message}`).join("; ")}` });
    return null;
  }
  const { children: _children, ...props } = result.data as Record<string, unknown>;
  const component: ReviewComponentNode = { type: "component", name, props, children };
  // Document order: parent before children, matching the audit's walk.
  context.components.splice(context.components.length - countComponents(children), 0, component);
  return component;
}

function countComponents(nodes: ReviewNode[]): number {
  let n = 0;
  for (const node of nodes) {
    if (node.type === "text") continue;
    if (node.type === "component") n += 1;
    n += countComponents(node.children);
  }
  return n;
}

function collectImportedLocals(mdast: MdastRoot, bindings: Record<string, unknown>, errors: ReviewDocumentTreeError[]): Record<string, unknown> {
  const esm = mdast.children.filter((node) => node.type === "mdxjsEsm");
  const locals: Record<string, unknown> = {};
  if (esm.length === 0) return locals;
  if (esm.length > 1) {
    errors.push({ line: nodeLine(esm[1]), message: `review.mdx may contain one import from "${DATA_IMPORT}" and no other import or export.` });
    return locals;
  }
  const program = (esm[0] as { data?: { estree?: Program } }).data?.estree;
  const statement = program?.body[0];
  if (!program || program.body.length !== 1 || !statement || statement.type !== "ImportDeclaration" || statement.source.value !== DATA_IMPORT) {
    errors.push({ line: nodeLine(esm[0]), message: `review.mdx may contain one import from "${DATA_IMPORT}" and no export.` });
    return locals;
  }
  for (const specifier of statement.specifiers) {
    if (specifier.type !== "ImportSpecifier") {
      errors.push({ line: nodeLine(esm[0]), message: `Import authored values by name: import { anchors } from "${DATA_IMPORT}".` });
      continue;
    }
    const imported = specifier.imported.type === "Identifier" ? specifier.imported.name : String(specifier.imported.value);
    if (!(imported in bindings)) {
      errors.push({ line: nodeLine(esm[0]), message: `data.ts does not export "${imported}".` });
      continue;
    }
    locals[specifier.local.name] = bindings[imported];
  }
  return locals;
}

// Attribute expressions are data paths into data.ts exports
// (`anchors.placeOrder`, `stores.db.tables.orders.status`, inline arrays and
// literals). They run with `new Function` in the CLI process at publish — the
// same trust boundary as importing data.ts itself. Nothing evaluated here
// ships to the app; only the resulting JSON does.
function expressionEvaluator(locals: Record<string, unknown>) {
  const names = Object.keys(locals);
  const values = names.map((name) => locals[name]);
  return (expression: string): unknown =>
    new Function(...names, `"use strict"; return (${expression});`)(...values);
}

function reactPropsFromHastProperties(properties: Record<string, unknown>): ReviewElementProps {
  const props: ReviewElementProps = {};
  for (const [key, value] of Object.entries(properties)) {
    if (value === undefined || value === null || value === false) continue;
    const info = find(html, key);
    const name = info.space ? (hastToReact[info.property] ?? info.property) : info.attribute;
    if (Array.isArray(value)) props[name] = value.join(info.commaSeparated ? ", " : " ");
    else if (typeof value === "string" || typeof value === "number" || typeof value === "boolean") props[name] = value;
  }
  return props;
}

function documentTitle(mdast: MdastRoot): string {
  const heading = mdast.children.find((node) => node.type === "heading" && node.depth === 1);
  if (!heading || heading.type !== "heading") return "";
  return heading.children.map((child) => ("value" in child ? child.value : "")).join("").trim();
}

function nodeLine(node: unknown): number | null {
  const line = (node as { position?: { start?: { line?: number } } })?.position?.start?.line;
  return typeof line === "number" ? line : null;
}
```

  ```ts
export async function buildReviewDocument(input: {
  ```

```ts
  const rangePeeks: ReviewPublishRangePeek[] = [];
  let peekCount = 0;
  let evidencePromise: Promise<ReviewPublishEvidenceTargets> | null = null;
  const evidence = () => {
    if (!input.prepareEvidence) throw new Error("Review source preparation is unavailable.");
    return (evidencePromise ??= input.prepareEvidence());
  };
  const resolveCodePeek = async (props: CodePeekProps, context?: CodePeekResolutionContext): Promise<CodePeekResolution> => {
    peekCount += 1;
    rangePeeks.push({ ...props, ...(context?.anchorId ? { anchorId: context.anchorId } : {}) });

      return { snapshot: { roots: [], resolved: {} }, diff: null } as unknown as CodePeekResolution; // copy the synthetic snapshot from review-publish-evaluate.ts:126-145 verbatim
    }
    const targets = await evidence();
    const primary = props.graph === "base" ? targets.base : targets.head;
    if (!primary) throw new Error(`Code peek range ${props.file}:${props.fromLine}-${props.toLine}: no ${props.graph ?? "head"} worktree.`);
    try {
      return await resolveReviewSourceRange({ rootPath: primary.sourceRootPath, root: { kind: "range", file: props.file, fromLine: props.fromLine, toLine: props.toLine } });
    } catch (error) {
      const message = `Code peek range ${props.file}:${props.fromLine}-${props.toLine}: ${error instanceof Error ? error.message : String(error)}`;
      failures.push(message);
      throw new Error(message);
    }
  };
  const session = createReviewDefinitionSession({ softwareMap: null, baseSoftwareMap: null, resolveCodePeek });
  session.begin();

  let exports: Record<string, unknown>;
  try {
    ({ exports } = await importReviewDataModule({ reviewDir: input.reviewDir, session }));
  } catch (error) {
    return { document: null, peekCount, rangePeeks, errors: [...failures, error instanceof Error ? error.message : String(error)], warnings: [] };
  }

  const source = await readFile(path.join(input.reviewDir, "review.mdx"), "utf8");
  const tree = buildReviewDocumentTree({ source, bindings: exports });
  for (const error of tree.errors) failures.push(`review.mdx${error.line ? `:${error.line}` : ""}: ${error.message}`);

  const callStackProps: CallStackDiffProps[] = [];
  const traceQuotes: PublishAuditTraceQuote[] = [];

    components: tree.components,

    collectCallStackDiff: (props) => callStackProps.push(props),
    collectTraceQuote: (quote) => traceQuotes.push(quote),
  });

  // from review-publish-evaluate.ts:225-322 here verbatim (they only read
  // callStackProps / traceQuotes / failures / evidence / input).
  const traceQuoteWarnings: string[] = [];

  const warnings = [...new Set([...session.diagnostics.map((d) => d.message), ...traceQuoteWarnings])];
  if (failures.length > 0) return { document: null, peekCount, rangePeeks, errors: failures, warnings };

    format: REVIEW_DOCUMENT_FORMAT,
    title: tree.title,
    routePath: input.routePath,
    sourcePath: "review.mdx",

    softwareModels,
  });
```

```tsx
import { type ComponentType, Fragment, type ReactNode, createElement } from "react";

import type { ReviewNode } from "../../src/review-document-data";
import type { reviewAuthoringComponents } from "./review-authoring-components";

export type ReviewDocumentComponents = typeof reviewAuthoringComponents &
  Record<string, ComponentType<never> | undefined>;

// The document is data: a tree of HTML elements, text, and registry
// components. Elements whose tag has an override (a, pre, h1) render through
// it; components render with the registry function itself so parents that
// inspect children by type (DatabaseLens, ReviewSection) keep working.
export function renderReviewNodes(
  nodes: ReviewNode[],
  components: ReviewDocumentComponents,
): ReactNode {
  return createElement(Fragment, null, ...nodes.map((node) => renderNode(node, components)));
}

function renderNode(node: ReviewNode, components: ReviewDocumentComponents): ReactNode {
  if (node.type === "text") return node.value;
  const children = node.children.map((child) => renderNode(child, components));
  if (node.type === "component") {
    const Component = components[node.name] as ComponentType<Record<string, unknown>>;
    return createElement(Component, node.props, ...children);
  }
  const override = components[node.tag] as ComponentType<Record<string, unknown>> | undefined;
  return createElement(override ?? node.tag, node.props, ...children);
}
```
